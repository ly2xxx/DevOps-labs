  --stack-name eksctl-eks-cluster-nodegroup-standard-workers \
  --query "StackEvents[?contains(ResourceStatus,'FAILED')].[LogicalResourceId,ResourceStatusReason]" \
  --output table

# instance types your account can use under the free tier, in this region
aws ec2 describe-instance-types --region us-west-2 \
  --filters Name=free-tier-eligible,Values=true \
  --query "InstanceTypes[].[InstanceType,VCpuInfo.DefaultVCpus,MemoryInfo.SizeInMiB]" \
  --output table

# which availability zones offer m7i-flex.large (your cluster uses 2b, 2c and 2d)

  --location-type availability-zone \
  --filters Name=instance-type,Values=m7i-flex.large \
  --query "InstanceTypeOfferings[].Location" --output text

#######################################################################################################
#Step 1: Set gp2 as the default StorageClass
kubectl patch storageclass gp2 -p '{"metadata": {"annotations":{"storageclass.kubernetes.io/is-default-class":"true"}}}'

#Step 2: Grant EBS permissions & install the aws-ebs-csi-driver add-on
#One-time creation
aws iam create-role \
  --role-name AmazonEKS_EBS_CSI_DriverRole \
  --assume-role-policy-document '{
    "Version": "2012-10-17",
    "Statement": [{
      "Effect": "Allow",
      "Principal": { "Service": "pods.eks.amazonaws.com" },
      "Action": ["sts:AssumeRole", "sts:TagSession"]
    }]
  }' 2>/dev/null || true
#One-time creation
aws iam attach-role-policy \
  --role-name AmazonEKS_EBS_CSI_DriverRole \
  --policy-arn arn:aws:iam::aws:policy/service-role/AmazonEBSCSIDriverPolicy

ACCOUNT_ID=$(aws sts get-caller-identity --query Account --output text)

aws eks create-pod-identity-association \
  --cluster-name eks-cluster \
  --region us-west-2 \
  --namespace kube-system \
  --service-account ebs-csi-controller-sa \
  --role-arn "arn:aws:iam::${ACCOUNT_ID}:role/AmazonEKS_EBS_CSI_DriverRole"

aws eks create-addon \
  --cluster-name eks-cluster \
  --addon-name aws-ebs-csi-driver \
  --region us-west-2

#One-time creation
#Step 3: Recreate the Postgres PVC so it picks up the new default StorageClass
kubectl -n ai-agent delete sts ai-agent-postgres
kubectl -n ai-agent delete pvc data-ai-agent-postgres-0
